Применить:
kubectl create namespace datalake
kubectl apply -f hdfs.yaml

kubectl apply -f spark-master.yaml
kubectl apply -f spark-worker.yaml


kubectl apply -f clickhouse-deployment.yaml
kubectl apply -f clickhouse-service.yaml


Проверить статус:
kubectl get pods -n datalake
kubectl get svc -n datalake


Создай папку (в одной папке с yaml):
mkdir -p ~/k8s-datalake/nginx
cd ~/k8s-datalake/nginx


kubectl apply -f nginx-sa.yaml
kubectl apply -f nginx-role.yaml
kubectl apply -f nginx-role-binding.yaml
kubectl apply -f nginx-ingress-service.yaml
kubectl apply -f nginx-ingress-controller.yaml

kubectl apply -f clickhouse-ingress.yaml
kubectl apply -f spark-ui-ingress.yaml

Удаление подов:
kubectl delete deployment spark-master spark-worker -n datalake
kubectl delete deployment nginx-ingress-controller -n datalake
kubectl delete pod -l app=nginx-ingress-controller -n datalake --force --grace-period=0
kubectl get all -n datalake | grep nginx
kubectl delete svc nginx-ingress -n datalake
